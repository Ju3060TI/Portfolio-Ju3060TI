// --- neron_core :: raymarch_sdf.glsl ---
// Signed distance field raymarcher with soft shadows
// and exponential fog. Compiled for GL_KHR_vulkan_glsl.

layout(location = 0) out vec4 fragColor;
layout(location = 0) in  vec2 vUV;

uniform vec2  uResolution;
uniform float uTime;
uniform vec3  uCamPos;

const int   MAX_STEPS = 128;
const float MAX_DIST  = 64.0;
const float SURF_EPS  = 0.0008;

float sdSphere(vec3 p, float r) {
    return length(p) - r;
}

float sdBox(vec3 p, vec3 b) {
    vec3 q = abs(p) - b;
    return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
}

float sdSmoothUnion(float a, float b, float k) {
    float h = clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0);
    return mix(b, a, h) - k * h * (1.0 - h);
}

float map(vec3 p) {
    float t = uTime * 0.35;
    vec3  q = p;
    q.xz = mat2(cos(t), -sin(t), sin(t), cos(t)) * q.xz;

    float sphere = sdSphere(q - vec3(0.0, 0.9, 0.0), 0.85);
    float boxA   = sdBox(p - vec3(-1.4, 0.4, 0.0), vec3(0.55));
    float boxB   = sdBox(p - vec3( 1.4, 0.4, 0.0), vec3(0.55));

    float d = sdSmoothUnion(sphere, boxA, 0.35);
    d = sdSmoothUnion(d, boxB, 0.35);

    // ground plane
    d = min(d, p.y + 0.02);
    return d;
}

vec3 calcNormal(vec3 p) {
    vec2 e = vec2(SURF_EPS, 0.0);
    return normalize(vec3(
        map(p + e.xyy) - map(p - e.xyy),
        map(p + e.yxy) - map(p - e.yxy),
        map(p + e.yyx) - map(p - e.yyx)
    ));
}

float softShadow(vec3 ro, vec3 rd, float mint, float maxt, float k) {
    float res = 1.0;
    float t   = mint;
    for (int i = 0; i < 48; i++) {
        float h = map(ro + rd * t);
        if (h < 0.001) return 0.0;
        res = min(res, k * h / t);
        t  += clamp(h, 0.02, 0.5);
        if (t > maxt) break;
    }
    return clamp(res, 0.0, 1.0);
}

vec3 render(vec3 ro, vec3 rd) {
    float t = 0.0;
    bool  hit = false;
    for (int i = 0; i < MAX_STEPS; i++) {
        vec3  p = ro + rd * t;
        float d = map(p);
        if (d < SURF_EPS) { hit = true; break; }
        t += d;
        if (t > MAX_DIST) break;
    }

    vec3 col = vec3(0.02, 0.02, 0.04);
    if (hit) {
        vec3 p = ro + rd * t;
        vec3 n = calcNormal(p);
        vec3 lightDir = normalize(vec3(0.6, 0.8, 0.4));
        float diff = clamp(dot(n, lightDir), 0.0, 1.0);
        float sh   = softShadow(p + n * 0.01, lightDir, 0.02, 12.0, 12.0);
        vec3  base = vec3(0.024, 0.714, 0.831); // #06b6d4 accent
        col  = base * diff * sh + vec3(0.02) * n.y;
    }

    // exponential fog
    float fog = 1.0 - exp(-0.05 * t * t);
    col = mix(col, vec3(0.008, 0.008, 0.016), fog);
    return col;
}

void main() {
    vec2 uv = (vUV * uResolution - 0.5 * uResolution) / uResolution.y;
    vec3 ro = uCamPos;
    vec3 rd = normalize(vec3(uv, 1.4));

    vec3 col = render(ro, rd);

    // hard tonemap — no gradients
    col = clamp(col, 0.0, 1.0);
    col = floor(col * 64.0) / 64.0; // posterize: brutalist banding

    fragColor = vec4(col, 1.0);
}
